01大数加法从低位逐位相加，记录进位后反转结果。＋
写法 1 · c++
#include <iostream>
#include <string>
#include <algorithm>
using namespace std;

int main() {
    string a, b;
    while (cin >> a >> b) {
        string res;
        int i = a.size() - 1;
        int j = b.size() - 1;
        int carry = 0;

        while (i >= 0 || j >= 0 || carry) {
            int x = (i >= 0) ? a[i] - '0' : 0;
            int y = (j >= 0) ? b[j] - '0' : 0;
            int sum = x + y + carry;

            res.push_back(sum % 10 + '0');
            carry = sum / 10;

            i--;
            j--;
        }

        reverse(res.begin(), res.end());
        cout << res << endl;
    }
    return 0;
}

04回文串从枚举子串改为由中心向两侧扩展。＋
写法 1 · C++
// 回文串（及子字符串）问题，但是这样解的时间复杂度较高，O(n^3)
class Solution {
public:
    int countSubstrings(string s) {
     int cnt = 0;
     for(int i=0; i<s.size(); i++){
        for(int j=i; j<s.size(); j++){
            string tmp =s.substr(i,j-i+1);
            if(iscircle(tmp)) cnt++;
        }
     }
     return cnt;
    }

private:
    bool iscircle(string tmp){
        int l = 0;
        int r = tmp.size()-1;
        while(l<r){
            if(tmp[l]!=tmp[r]) return false;
            l++;
            r--;
        }
        return true;
    }
};
写法 2 · C++
/*
利用回文串的特性来解：从中心向两边扩展，如果两边字符相同，就是回文
【任何回文串，都有一个中心。我们枚举了所有可能的中心，并且从每个中心尽可能向两边扩展，所以每个回文串都会被找到一次。】
*/
class Solution {
public:
    int countSubstrings(string s) {
     int cnt = 0;
     for(int i=0; i<s.size(); i++){
        cnt += subcnt(s,i,i);
        cnt += subcnt(s,i,i+1);
     }
     return cnt;
    }

private:
    int subcnt(string tmp, int l, int r){
        int cnt = 0;
        while(l>=0 && r<tmp.size()){
            if(tmp[l]==tmp[r]){
                cnt++;
                l--;
                r++;
            }else{
                break;
            }
        }
        return cnt;
    }
};

06移动窗口固定长度窗口移出左端、加入右端，维护区间和。＋
写法 1 · C++
class Solution {
public:
    double findMaxAverage(vector<int>& nums, int k) {
        // 求最大sum即可
        int sum = 0;
        int maxx = 0;
        // 先算出第一个窗口的sum
        for(int i=0; i<k; i++){
            sum += nums[i];
        }
        maxx = sum;
        //依次减去前面的值，再加上后面的值
        for(int j=k; j<nums.size(); j++){
            sum += nums[j];
            sum -= nums[j-k];
            maxx = max(sum, maxx);
        }
        return (double)maxx/k;
    }
};
07二分最基础题目（递归）猜数字题中的递归二分与边界收缩。＋
写法 1 · C++
/**
 * Forward declaration of guess API.
 * @param  num   your guess
 * @return 	     -1 if num is higher than the picked number
 *			      1 if num is lower than the picked number
 *               otherwise return 0
 * int guess(int num);
 */

class Solution {
public:
    int guessNumber(int n) {
        int lnum = 1;
        int rnum = n;
        // 等于的情况，直接选中pick
        if(guess(lnum)==0) return lnum;
        if(guess(rnum)==0) return rnum;
        int pick = jugnumber(lnum, rnum);
        return pick;
    }
private:
    int jugnumber(int ln, int rn){
        // 避免越界问题
        int mid = ln + (rn-ln)/2;
        int guessmid = guess(mid);
        // 实际数字更大
        if(guessmid>0){
            return jugnumber(mid+1, rn);
        }
        // 实际数字更小
        if(guessmid<0){
            return jugnumber(ln, mid-1);
        }
        // 等于的情况，直接返回
        return mid;
    }
};

            // 找到了则说明比这个下标更大的下标所代表的值都是满足的
            tmp[i] =  (posize - pos);
        }
        return tmp;
    }

private:
    int successfulNode(int spells, vector<int>& potions, long long success, int left, int right){
        // 递归出口，在已经二分搜索完毕之后发现全都符合要求，那就返回输入的左值0，让外部知道全都满足要求
        if (left > right) return left;
        int mid = left + (right-left)/2;
        // 大于等于最小需求值，在左半边继续寻找更小但满足需求的值
        if( success<=(long long)spells*potions[mid] ) return successfulNode(spells, potions, success, left, mid-1);
        // 其余情况则在右半边 继续进行迭代搜索
        return successfulNode(spells, potions, success, mid+1, right);
    }
};

class Solution {
public:
    vector<int> successfulPairs(vector<int>& spells, vector<int>& potions, long long success) {
        int spsize = spells.size();
        int posize = potions.size();
        vector<int> tmp(spsize);

        sort(potions.begin(), potions.end());

        for (int i = 0; i < spsize; i++) {
            // 找到第一个满足 spells[i] * potions[pos] >= success 的位置
            int pos = lowerBound(spells[i], potions, success, 0, posize - 1);
            // 如果 pos == posize，说明没有满足条件的
            tmp[i] = (pos == posize) ? 0 : (posize - pos);
        }
        return tmp;
    }

private:
    // 返回第一个满足 spells * potions[mid] >= success 的下标
    // 如果都不满足，返回 potions.size()
    int lowerBound(int spell, vector<int>& potions, long long success, int left, int right) {
        int posize = potions.size();
        while (left <= right) {
            int mid = left + (right - left) / 2;
            if ((long long)spell * potions[mid] >= success) {
                // mid 满足，但左边可能还有更小的满足，往左找
                right = mid - 1;
            } else {
                // mid 不满足，往右找
                left = mid + 1;
            }
        }
        /*
        找到最后发现全都满足条件，且left就是第一个满足条件的位置，所以返回left位置
        */
        return left;
    }
};

13右视角的节点值（递归）先右后左递归，每层只记录首次访问的节点。＋
写法 1 · C++
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    vector<int> rightSideView(TreeNode* root) {
        vector<int> rtemp;
        // 递归
        rightdfs(root, 0, rtemp);
        return rtemp;
    }

private:
    void rightdfs( TreeNode* node, int height, vector<int>& rc ){
        if(!node) return;
        if(height == rc.size()) rc.push_back(node->val);
        // 按照 右子树-》左子树的 顺序递归
        rightdfs(node->right, height+1, rc);
        rightdfs(node->left, height+1, rc);
    }
};
14对称树（二叉树、递归）成对比较镜像节点及其左右子树。＋
写法 1 · C++
// 关键在于要对于对称的条件进行梳理
// 两个都为空 → 对称
// 只有一个为空 → 不对称
// 两个都不为空，则需要同时满足：
//1、left->val == right->val
//2、left->left 和 right->right 互为镜像
//3、left->right 和 right->left 互为镜像
// 并且对于实际对比的对象也要理清楚：left->val == right->val
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    bool isSymmetric(TreeNode* root) {
        if(!root) return true;
        return ifmirror(root->left, root->right);
    }

private:
    bool ifmirror(TreeNode* left, TreeNode* right){
        if(!left && !right) return true;
        if(!left || !right) return false;
        bool f = false;
        if((left->val == right->val)
            && ifmirror(left->left,right->right)
            && ifmirror(left->right,right->left)){
                f = true;
            }
        return f;
    }
};

        root->right = tmp;

        invertTree(root->left);
        invertTree(root->right);

        return root;
    }

/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}

 * };
 */
class Solution {
public:
    int maxDepth(TreeNode* root) {
        if(!root) return 0;
        // 根节点也算是高度为1
        int depth = 1;

        int ld = maxDepth(root->left);
        int rd = maxDepth(root->right);

        depth += max(ld,rd);

        return depth;
    }
};
17两个链表的相交点比较节点地址；记录了双重遍历和哈希集合两种写法。＋
写法 1 · C++
// 注意比较的对象是节点地址，即：函数传递的参数指针的值
/*
本题还有更加简单的方式：
1、数学推理（将两个链表实际可到达的位置互相桥接一下，可以发现这两个链表实际实际走的路程一样）【实际不容易想到】
设 A 的长度为 a，B 的长度为 b，相交部分长度为 c，A 独有部分 a-c，B 独有部分 b-c。
pA 走的路程：a + (b - c)
pB 走的路程：b + (a - c)
两者总路程相等，都是 a + b - c。所以它们会在相交点相遇。
2、哈希集合【这个可以多熟悉，比较容易想到，但是要对哈希集合特性比较熟悉】
*/
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode(int x) : val(x), next(NULL) {}
 * };
 */
class Solution {
public:
    ListNode *getIntersectionNode(ListNode *headA, ListNode *headB) {
        if(!headA || !headB) return nullptr;
        bool f = false;
        ListNode * fnode = nullptr;
        while(headA){
            ListNode* hb = headB;
            while(hb){
                if( headA == hb ){
                    f = true;
                    fnode = headA;
                    break;
                }
                hb = hb->next;
            }
            if(!f) headA = headA->next;
            else break;
        }
        return fnode;
    }
};

18移除链表倒数第n个节点从计算长度，推进到哑节点与快慢指针。＋
写法 1 · C++
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    ListNode* removeNthFromEnd(ListNode* head, int n) {
        if(!head) return nullptr;
        ListNode* tmp = head;
        int sizel = 0;
        while(tmp){
            tmp = tmp->next;
            sizel++;
        }
        tmp = head;

        // 如果要删除的是头节点
        if (n == sizel) {
            ListNode* newHead = head->next;
            delete head;   // 可选
            return newHead;
        }

        int ind = 0;
        while(tmp){
            if(ind == sizel-n-1){
                ListNode* t = tmp->next;
                tmp->next = t->next;
                delete t;
                break;
            }
            ind++;
            tmp=tmp->next;
        }
        return head;
    }
};

/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}

        ListNode* dump = new ListNode(0);

        ListNode* tmp = slow->next;
        slow->next = slow->next->next;
        delete tmp;

        ListNode* cur = dump->next;
        delete dump;
        return cur;
    }
};

 * struct ListNode {

 *     ListNode *next;

21跳跃游戏(贪心)维护当前可到达的最远下标。＋
写法 1 · C++
// 注意审题，以及对于贪心的理解：每次都选择当前这一步最好的情况
class Solution {
public:
    bool canJump(vector<int>& nums) {
        bool f = false;
        int sizen = nums.size();
        int maxr = 0;
        for(int i=0; i<sizen; i++){
            /*
            最大范围步数随着更新仍然小于当前移动的下标，说明已经不可能到达了
            */
            if(i>maxr){
                f=false;
                break;
            }
            // 每次寻找能够到达的最大的范围步数
            /*
            这一步代表的是基于当前位置的下一步能够到达的所有位置依次进行比较是否可以更新最远的下标值
            */
            maxr = max(maxr, i+nums[i]);
            if(maxr>=sizen-1){
                f = true;
                break;
            }
        }
        return f;
    }
};
22跳跃游戏Ⅱ（贪心）维护当前跳跃边界和下一跳可达范围。＋
写法 1 · C++
// 重点在于只需要维护“范围”，不关心中间点是哪些
class Solution {
public:
    int jump(vector<int>& nums) {
         int sizen = nums.size();
         if(sizen<=1) return 0;
         // 当前这一跳能够到达的最远位置下标
         int curend = 0;
         // 下一跳能够到达的最远位置下标
         int furend = 0;
         // 跳跃次数
         int jump = 0;
         for(int i=0; i<sizen; i++){
            // 下一跳能够到达的最远位置下标
            furend = max(furend, i+nums[i]);
            // 到达当前这一次跳跃的边界时，必须要跳才行
            if(i == curend){
                jump++;
                // 跳到下一跳能够到达的最远的位置
                curend = furend;
                // 到达最远下标时，就跳出循环
                if(curend>=sizen-1) break;
            }
         }
         return jump;
    }
};
23气球射击(贪心)按右端点排序，用尽量少的箭覆盖区间。＋
写法 1 · C++
// 区间贪心，主要在于先要进行排序，并且将实际每一次贪心的规则进行总结
class Solution {
public:
    int findMinArrowShots(vector<vector<int>>& points) {
        if(points.empty()) return 0;
        // 先按照右端点的下标进行升序排列
        sort(points.begin(), points.end(), [](const vector<int>& a, const vector<int>& b){ return a[1]<b[1]; });

        // 之后按照贪心的思想，将每一个气球end位置依次初始化为每一个开始的位置
        int sizep = points.size();
        int sum = 1;
        int nr = points[0][1];
        for(int i=1; i<sizep; i++){
            if(points[i][0]>nr){
                sum++;
                nr = points[i][1];
            }
        }
        return sum;
    }
};
24不重叠的时间间隔（贪心）按右端点排序，先统计能保留的区间。＋
写法 1 · C++
// 主要问题在于要搞清楚当前计数的是什么（上一道题其实求的是最小重叠个数，即重叠的部分尽可能多的情况下，统计总共有多少重叠），而这道题其实求的是相反的值：即不重叠的个数
class Solution {
public:
    int eraseOverlapIntervals(vector<vector<int>>& intervals) {
        if(intervals.empty()) return 0;
        int sizei = intervals.size();
        sort(intervals.begin(), intervals.end(), [](const vector<int>& a, const vector<int>& b){
            return b[1]>a[1];
        });
        int nr = intervals[0][1];
        int keep = 1;
        for(int i=1; i<sizei; i++){
            // 不重叠则记录不重叠的个数
            if(intervals[i][0]>=nr){
                keep++;
                nr = intervals[i][1];
            }
        }
        return sizei - keep;
    }
};
