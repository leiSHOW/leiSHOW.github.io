记LC 76 · 最小窗口子串2026-10-15 的练习与复盘＋

主要考察对于哈希表（unordered_map<char, int>）的使用

这道题不是找连续模式，而是找“覆盖”：

  • 不要求 t 作为连续子串出现。
  • 不要求字符顺序。
  • 只要求窗口内每个字符的数量都不少于 t 中的数量。

并且对于这两个哈希表的应用也很关键：

  • need：记录 t 中每个字符需要多少个。
  • window：记录当前窗口 [left, right) 中每个字符有多少个。

因为对于右边界的自增是： 先取字符，再 right++ 。

所以其实实际right是在right+1的位置

最终更新答案的时候窗口长度是 right - left

并且要注意 s.substr(start, len)是从start下标开始的len个字符

写法 1 · C++
class Solution {
public:
    string minWindow(string s, string t) {
        unordered_map<char, int> need, window;
        // 将匹配串放入到哈希表中，用于统计t中每个字符的需求数量
        for(char c : t) need[c]++;
        // 左右边界[left, right)
        int left = 0, right = 0;
        // 窗口中已经满足需求的字符种类数目
        int valid = 0;
        // 最小覆盖子串的起始位置和长度
        int start = 0, len = INT_MAX;

        while(right < s.size()){
            char c = s[right];
            // 右移窗口
            right++;
            // 如果当前c在需求窗口中就加入顺序子窗口
            if(need.count(c)){
                window[c]++;
                // 如果当前字符满足需求数目，就说明已有一个字符是满足在这个顺序子窗口中的
                if(window[c]==need[c]) valid++;
            }
            // 此时，说明当前窗口已经覆盖了t的所有字符，尝试去收缩左边界
            while(valid == need.size()){
                // 更新当前答案的起始位置和长度
                if(right-left <len){
                    start = left;
                    len = right - left;
                }
                // 从当前起始位置开始尝试去收缩左边界
                char d = s[left];
                left++;
                // 如果d在当前需求的字符中，删除满足需求的值以及需求子窗口的情况
                if(need.count(d)){
                    if(window[d]==need[d]) valid--;
                    window[d]--;
                }
            }
        }
        return len == INT_MAX ? "" : s.substr(start, len);
    }
};

记LC 42 · 收集雨水2026-10-08 的练习与复盘＋

找规律，以及贪心

这道题主要是想象每个柱子上方都有一个“水槽”，能接多少水取决于它左右两边最高的柱子中较矮的那个，再减去它自身的高度。

核心公式：

写法 1 · C++
位置 i 的接水量 = max(0, min(左边最高, 右边最高) - height[i])

为什么是 min？因为水会从较矮的一边流走，所以水位最高只能到较矮的那根柱子。

但由于这道题有时间限制，常规暴力解会超时，所以可以先提前用两个数组存起来：

  • maxl[i]：从 0 到 i 的最高柱子高度。
  • maxr[i]：从 i 到 n-1 的最高柱子高度。
写法 2 · C++
class Solution {
public:
    int trap(vector<int>& height) {
        if(height.empty()) return 0;
        int sizeh = height.size();
        vector<int> maxl(sizeh), maxr(sizeh);
        int ans = 0;

        maxl[0] = height[0];
        for(int i=1; i<sizeh; i++){
            maxl[i] = max(maxl[i-1], height[i]);
        }

        maxr[sizeh-1] = height[sizeh-1];
        for(int i=sizeh-2; i>-1; i--){
            maxr[i] = max(maxr[i+1], height[i]);
        }

        for(int i=1; i<sizeh-1; i++){
            ans += min(maxl[i],maxr[i]) - height[i];
        }
        return ans;
    }
};

这道题由于数组是无序的，并且解题是与下标有关的，也不能按照先排序再求解的方式进行，所以可以根据这道题查找第一个满足条件的下标的值的特性，使用栈的入栈、出栈特性来解题：

写法 2 · C++
class Solution {
public:
    vector<int> dailyTemperatures(vector<int>& temperatures) {
        int sizet = temperatures.size();
        vector<int> ans(sizet,0);
        stack<int> st;

        for(int i=0; i<sizet; i++){
            while(!st.empty() && temperatures[i] > temperatures[st.top()]){
                int pre = st.top();
                st.pop();
                ans[pre] = i - pre;
            }
            st.push(i);
        }
        return ans;
    }
};

记LC 34 · 排序数组中查找元素第一个和最后一个2026-09-30 的练习与复盘＋

二分（左右边界查找）

写法 1 · C++
// 注意左边界和右边界的查找
class Solution {
public:
    vector<int> searchRange(vector<int>& nums, int target) {
        if(nums.empty()) return {-1,-1};
        vector<int> result;
        int f = leftcheck(nums,target);
        int l = rightcheck(nums,target);
        result.push_back(f);
        result.push_back(l);
        if (l >= nums.size() || nums[l] != target) {
            return {-1, -1};
        }
        return result;
    }

private:
    int leftcheck(vector<int>& nums, int target){
        int sizen = nums.size();
        int l = 0;
        int r = sizen-1;
        while(l<=r){
            int mid = l + (r-l)/2;
            if(nums[mid]>=target){
            // mid 可能不满足“第一个”的条件（即使 nums[mid] == target，左边可能还有更早的），所以要排除 mid，继续往左找。
                r = mid -1;
            }else{
                l = mid+1;
            }
        }
        return l;
    }

    int rightcheck(vector<int>& nums, int target){
        int sizen = nums.size();
        int l = 0;
        int r = sizen-1;
        while(l<=r){
            int mid = l + (r-l)/2;
            if(nums[mid]<=target){
            // 同理，mid 可能不满足“最后一个”的条件（即使 nums[mid] == target，右边可能还有更晚的），所以要排除 mid，继续往右找。
                l = mid +1;
            }else{
                r = mid-1;
            }
        }
        return r;
    }
};

并且对于为什么需要特判左边界，主要是因为：

写法 2 · C++
int leftBound(vector<int>& nums, int target) {
    int l = 0, r = nums.size() - 1;
    while (l <= r) {
        int mid = l + (r - l) / 2;
        if (nums[mid] >= target) {
            r = mid - 1;   // 往左找
        } else {
            l = mid + 1;
        }
    }
    return l;   // l 是第一个 >= target 的位置
}

l 的取值范围是 [0, nums.size()] 。

  • 如果所有元素都 < target，l 会一路增加到 nums.size()（越界）。
  • 如果找到了 target，l 就是第一个 == target 的位置。

而对于右边界，其实也是一样的：

写法 3 · C++
int rightBound(vector<int>& nums, int target) {
    int l = 0, r = nums.size() - 1;
    while (l <= r) {
        int mid = l + (r - l) / 2;
        if (nums[mid] <= target) {
            l = mid + 1;   // 往右找
        } else {
            r = mid - 1;
        }
    }
    return r;   // r 是最后一个 <= target 的位置
}

r 的取值范围是 [-1, nums.size()-1] 。

  • 如果所有元素都 > target，r 会一路减小到 -1（越界）。
  • 如果找到了 target，r 就是最后一个 == target 的位置。

但是由于核心逻辑： left 的检查已经能判断 target 是否存在。

写法 4 · C++
if (left >= nums.size() || nums[left] != target) {
    return {-1, -1};
}

这两种情况覆盖了所有“target 不存在”的可能：

情况 left 的值 nums[left] 判断结果
所有元素都 < target nums.size() 越界 left >= size ✅ 返回 [-1,-1]
target 不存在，但有些元素比它大 第一个 > target 的位置 != target 返回 [-1,-1]
target 存在 第一个 == target 的位置 == target 通过检查

即：一旦通过左边界的检查，说明 target 一定存在。

class Solution {
public:
    int searchInsert(vector<int>& nums, int target) {
        int l = 0, r = nums.size();   // 注意：r = size，不是 size-1
        while (l < r) {
            int mid = l + (r - l) / 2;
            if (nums[mid] < target) {
                l = mid + 1;
            } else {
                r = mid;
            }
        }
        return l;   // l 就是插入位置
    }
};

这样处理能够综合考虑到在“空数组”以及“数组中所有元素都小于target”情况下最后一次l的指向下标问题：

  • 区间是 [l, r)，当 l == r 时区间为空，循环结束。
  • l 始终指向第一个 >= target 的位置，如果所有元素都 < target，l 会一直增加到 r = sizen，正好是插入到末尾的位置。
  • 不需要任何特判，空数组也能正确处理（l=0, r=0，返回 0）。

写法 1 · c++
#include <iostream>
#include <string>
#include <algorithm>
using namespace std;

int main() {
    string a, b;
    while (cin >> a >> b) {
        string res;
        int i = a.size() - 1;
        int j = b.size() - 1;
        int carry = 0;

        while (i >= 0 || j >= 0 || carry) {
            int x = (i >= 0) ? a[i] - '0' : 0;
            int y = (j >= 0) ? b[j] - '0' : 0;
            int sum = x + y + carry;

            res.push_back(sum % 10 + '0');
            carry = sum / 10;

            i--;
            j--;
        }

        reverse(res.begin(), res.end());
        cout << res << endl;
    }
    return 0;
}

04回文串LC 647 ↗从枚举子串改为由中心向两侧扩展。＋
写法 1 · C++
// 回文串（及子字符串）问题，但是这样解的时间复杂度较高，O(n^3)
class Solution {
public:
    int countSubstrings(string s) {
     int cnt = 0;
     for(int i=0; i<s.size(); i++){
        for(int j=i; j<s.size(); j++){
            string tmp =s.substr(i,j-i+1);
            if(iscircle(tmp)) cnt++;
        }
     }
     return cnt;
    }

private:
    bool iscircle(string tmp){
        int l = 0;
        int r = tmp.size()-1;
        while(l<r){
            if(tmp[l]!=tmp[r]) return false;
            l++;
            r--;
        }
        return true;
    }
};
写法 2 · C++
/*
利用回文串的特性来解：从中心向两边扩展，如果两边字符相同，就是回文
【任何回文串，都有一个中心。我们枚举了所有可能的中心，并且从每个中心尽可能向两边扩展，所以每个回文串都会被找到一次。】
*/
class Solution {
public:
    int countSubstrings(string s) {
     int cnt = 0;
     for(int i=0; i<s.size(); i++){
        cnt += subcnt(s,i,i);
        cnt += subcnt(s,i,i+1);
     }
     return cnt;
    }

private:
    int subcnt(string tmp, int l, int r){
        int cnt = 0;
        while(l>=0 && r<tmp.size()){
            if(tmp[l]==tmp[r]){
                cnt++;
                l--;
                r++;
            }else{
                break;
            }
        }
        return cnt;
    }
};

06移动窗口LC 643 ↗固定长度窗口移出左端、加入右端，维护区间和。＋
写法 1 · C++
class Solution {
public:
    double findMaxAverage(vector<int>& nums, int k) {
        // 求最大sum即可
        int sum = 0;
        int maxx = 0;
        // 先算出第一个窗口的sum
        for(int i=0; i<k; i++){
            sum += nums[i];
        }
        maxx = sum;
        //依次减去前面的值，再加上后面的值
        for(int j=k; j<nums.size(); j++){
            sum += nums[j];
            sum -= nums[j-k];
            maxx = max(sum, maxx);
        }
        return (double)maxx/k;
    }
};
07二分最基础题目（递归）LC 374 ↗猜数字题中的递归二分与边界收缩。＋
写法 1 · C++
/**
 * Forward declaration of guess API.
 * @param  num   your guess
 * @return 	     -1 if num is higher than the picked number
 *			      1 if num is lower than the picked number
 *               otherwise return 0
 * int guess(int num);
 */

class Solution {
public:
    int guessNumber(int n) {
        int lnum = 1;
        int rnum = n;
        // 等于的情况，直接选中pick
        if(guess(lnum)==0) return lnum;
        if(guess(rnum)==0) return rnum;
        int pick = jugnumber(lnum, rnum);
        return pick;
    }
private:
    int jugnumber(int ln, int rn){
        // 避免越界问题
        int mid = ln + (rn-ln)/2;
        int guessmid = guess(mid);
        // 实际数字更大
        if(guessmid>0){
            return jugnumber(mid+1, rn);
        }
        // 实际数字更小
        if(guessmid<0){
            return jugnumber(ln, mid-1);
        }
        // 等于的情况，直接返回
        return mid;
    }
};

public:
    vector<int> successfulPairs(vector<int>& spells, vector<int>& potions, long long success) {

        int posize = potions.size();
        vector<int> tmp(spsize);
        // 对potions先从小到大进行排序

        for(int i=0; i<spsize; i++){
            int pos = successfulNode(spells[i], potions, success, 0, potions.size()-1);
            // 找到了则说明比这个下标更大的下标所代表的值都是满足的
            tmp[i] =  (posize - pos);
        }
        return tmp;
    }

private:
    int successfulNode(int spells, vector<int>& potions, long long success, int left, int right){
        // 递归出口，在已经二分搜索完毕之后发现全都符合要求，那就返回输入的左值0，让外部知道全都满足要求
        if (left > right) return left;
        int mid = left + (right-left)/2;
        // 大于等于最小需求值，在左半边继续寻找更小但满足需求的值
        if( success<=(long long)spells*potions[mid] ) return successfulNode(spells, potions, success, left, mid-1);
        // 其余情况则在右半边 继续进行迭代搜索
        return successfulNode(spells, potions, success, mid+1, right);
    }
};

            tmp[i] = (pos == posize) ? 0 : (posize - pos);

        return tmp;

    // 如果都不满足，返回 potions.size()

        int posize = potions.size();
        while (left <= right) {
            int mid = left + (right - left) / 2;
            if ((long long)spell * potions[mid] >= success) {
                // mid 满足，但左边可能还有更小的满足，往左找
                right = mid - 1;
            } else {
                // mid 不满足，往右找
                left = mid + 1;
            }
        }
        /*
        找到最后发现全都满足条件，且left就是第一个满足条件的位置，所以返回left位置
        */
        return left;
    }

// 注意对于特殊情况的处理：只有一个节点时，应该怎么处理
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    ListNode* middleNode(ListNode* head) {
        if(!head) return nullptr;
        if(head && head->next==nullptr) return head;
        ListNode* tmp = nullptr;
        ListNode* mid = nullptr;
        int sizel = 0;
        tmp = head;
        while(tmp){
            sizel++;
            tmp = tmp->next;
        }
        tmp = head;
        int i = 0;
        while(tmp){
                if(i==sizel/2-1){
                    mid = tmp->next;
                    break;
                }
            i++;
            tmp = tmp->next;
        }
        return mid;
    }
};

可以考虑用“快慢指针”来做，思路更简单并且逻辑更清晰

写法 2 · C++
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    ListNode* middleNode(ListNode* head) {
        ListNode* slow = head;
        ListNode* fast = head;
        // 快指针每次往前两步，慢指针每次往前一步
        while(fast && fast->next){
            slow = slow->next;
            // 奇数个节点，正好fast->next为空时，slow指向中间节点
            // 偶数个节点，正好fast为空时，slow指向中间节点的第二个
            fast = fast->next->next;
        }
        return slow;
    }
};
22翻转链表LC 206 ↗用前驱、当前和下一节点三个指针原地改链。＋
写法 1 · C++
/*
注意pre和cur开始的指向，开始的指向决定了是否能在后续进行持续推进
*/
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    ListNode* reverseList(ListNode* head) {
        if(!head) return nullptr;
        ListNode* pre = nullptr;
        ListNode* cur = head;
        while(cur){
            ListNode* tmp = cur->next;
            cur ->next = pre;
            pre = cur;
            cur = tmp;
        }
        return pre;
    }
};

23跳跃游戏(贪心)LC 55 ↗维护当前可到达的最远下标。＋
写法 1 · C++
// 注意审题，以及对于贪心的理解：每次都选择当前这一步最好的情况
class Solution {
public:
    bool canJump(vector<int>& nums) {
        bool f = false;
        int sizen = nums.size();
        int maxr = 0;
        for(int i=0; i<sizen; i++){
            /*
            最大范围步数随着更新仍然小于当前移动的下标，说明已经不可能到达了
            */
            if(i>maxr){
                f=false;
                break;
            }
            // 每次寻找能够到达的最大的范围步数
            /*
            这一步代表的是基于当前位置的下一步能够到达的所有位置依次进行比较是否可以更新最远的下标值
            */
            maxr = max(maxr, i+nums[i]);
            if(maxr>=sizen-1){
                f = true;
                break;
            }
        }
        return f;
    }
};
24跳跃游戏Ⅱ（贪心）LC 45 ↗维护当前跳跃边界和下一跳可达范围。＋
写法 1 · C++
// 重点在于只需要维护“范围”，不关心中间点是哪些
class Solution {
public:
    int jump(vector<int>& nums) {
         int sizen = nums.size();
         if(sizen<=1) return 0;
         // 当前这一跳能够到达的最远位置下标
         int curend = 0;
         // 下一跳能够到达的最远位置下标
         int furend = 0;
         // 跳跃次数
         int jump = 0;
         for(int i=0; i<sizen; i++){
            // 下一跳能够到达的最远位置下标
            furend = max(furend, i+nums[i]);
            // 到达当前这一次跳跃的边界时，必须要跳才行
            if(i == curend){
                jump++;
                // 跳到下一跳能够到达的最远的位置
                curend = furend;
                // 到达最远下标时，就跳出循环
                if(curend>=sizen-1) break;
            }
         }
         return jump;
    }
};
25气球射击(贪心)LC 452 ↗按右端点排序，用尽量少的箭覆盖区间。＋
写法 1 · C++
// 区间贪心，主要在于先要进行排序，并且将实际每一次贪心的规则进行总结
class Solution {
public:
    int findMinArrowShots(vector<vector<int>>& points) {
        if(points.empty()) return 0;
        // 先按照右端点的下标进行升序排列
        sort(points.begin(), points.end(), [](const vector<int>& a, const vector<int>& b){ return a[1]<b[1]; });

        // 之后按照贪心的思想，将每一个气球end位置依次初始化为每一个开始的位置
        int sizep = points.size();
        int sum = 1;
        int nr = points[0][1];
        for(int i=1; i<sizep; i++){
            if(points[i][0]>nr){
                sum++;
                nr = points[i][1];
            }
        }
        return sum;
    }
};
26不重叠的时间间隔（贪心）LC 435 ↗按右端点排序，先统计能保留的区间。＋
写法 1 · C++
// 主要问题在于要搞清楚当前计数的是什么（上一道题其实求的是最小重叠个数，即重叠的部分尽可能多的情况下，统计总共有多少重叠），而这道题其实求的是相反的值：即不重叠的个数
class Solution {
public:
    int eraseOverlapIntervals(vector<vector<int>>& intervals) {
        if(intervals.empty()) return 0;
        int sizei = intervals.size();
        sort(intervals.begin(), intervals.end(), [](const vector<int>& a, const vector<int>& b){
            return b[1]>a[1];
        });
        int nr = intervals[0][1];
        int keep = 1;
        for(int i=1; i<sizei; i++){
            // 不重叠则记录不重叠的个数
            if(intervals[i][0]>=nr){
                keep++;
                nr = intervals[i][1];
            }
        }
        return sizei - keep;
    }
};
